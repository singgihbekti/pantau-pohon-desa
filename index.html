const FOLDER_ID = '13OamEoEjC5espwU7aGinGUPchkATECK-';
const SHEET_NAME = 'Log_Pohon';

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = getOrCreateSheet_(ss, SHEET_NAME);
    const values = sheet.getDataRange().getValues();

    if (values.length <= 1) {
      return jsonResponse_({ ok: true, data: [] });
    }

    const headers = values[0].map(function (h) {
      return String(h || '').trim();
    });

    const rows = [];
    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      const hasContent = row.some(function (cell) {
        return cell !== '' && cell !== null;
      });
      if (!hasContent) continue;

      const obj = {};
      for (let j = 0; j < headers.length; j++) {
        obj[headers[j]] = row[j];
      }
      rows.push(obj);
    }

    return jsonResponse_({ ok: true, data: rows });
  } catch (err) {
    return jsonResponse_({ ok: false, error: err.message });
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse_({ ok: false, error: 'Payload kosong' });
    }

    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = getOrCreateSheet_(ss, SHEET_NAME);

    let fotoUrl = String(payload.fotoUrl || payload.Foto_URL || '').trim();

    if (payload.fotoBase64 && String(payload.fotoBase64).indexOf('base64,') !== -1) {
      fotoUrl = saveBase64ToDrive_(
        payload.fotoBase64,
        payload.kodePohon || payload.Kode_Pohon || 'POHON',
        payload.namaFile || 'foto.jpg'
      );
    }

    const now = new Date();
    const bulanLaporan =
      payload.bulanLaporan ||
      payload.Bulan_Laporan ||
      Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM');

    const rowData = [
      now,
      String(payload.kodePohon || payload.Kode_Pohon || '').trim().toUpperCase(),
      String(payload.namaWarga || payload.Nama_Warga || '').trim(),
      String(payload.wilayah || payload.Wilayah || '').trim(),
      String(payload.jenisPohon || payload.Jenis_Pohon || '').trim(),
      Number(payload.latitude || payload.Latitude || 0) || '',
      Number(payload.longitude || payload.Longitude || 0) || '',
      Number(payload.tinggiCm || payload.Tinggi_cm || 0) || '',
      String(payload.kualitas || payload.Kualitas || 'Baik').trim(),
      String(payload.catatan || payload.Catatan || '').trim(),
      fotoUrl,
      String(bulanLaporan).trim()
    ];

    sheet.appendRow(rowData);

    return jsonResponse_({
      ok: true,
      message: 'Data berhasil disimpan',
      fotoUrl: fotoUrl
    });
  } catch (err) {
    return jsonResponse_({ ok: false, error: err.message });
  }
}

function saveBase64ToDrive_(base64Data, kodePohon, namaFile) {
  const parts = String(base64Data).split(',');
  const headerPart = parts[0] || '';
  const rawBase64 = parts[1] || parts[0];

  const mimeMatch = headerPart.match(/data:(.*?);base64/);
  const mimeType = mimeMatch ? mimeMatch[1] : 'image/jpeg';
  const ext = mimeType.indexOf('png') !== -1 ? 'png' : 'jpg';

  const decoded = Utilities.base64Decode(rawBase64);
  const cleanCode = String(kodePohon || 'POHON').replace(/[^a-zA-Z0-9_-]/g, '_');
  const timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd_HHmmss');
  const safeName = cleanCode + '_' + timestamp + '.' + ext;

  const blob = Utilities.newBlob(decoded, mimeType, safeName);
  const folder = DriveApp.getFolderById(FOLDER_ID);
  const file = folder.createFile(blob);

  try {
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (e) {}

  return 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w800';
}

function getOrCreateSheet_(ss, sheetName) {
  let sheet = ss.getSheetByName(sheetName);
  const headers = [
    'Timestamp',
    'Kode_Pohon',
    'Nama_Warga',
    'Wilayah',
    'Jenis_Pohon',
    'Latitude',
    'Longitude',
    'Tinggi_cm',
    'Kualitas',
    'Catatan',
    'Foto_URL',
    'Bulan_Laporan'
  ];

  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(headers);
  } else if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
  }

  return sheet;
}

function jsonResponse_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
