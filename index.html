Mengapa Tanpa Login Tetap Terlacak?

  • 1. Setiap pohon memiliki Kode Pohon unik (mis. PHN-001) yang dapat dicetak di label/papan kecil di pohon atau dicatat warga.
  • 2. Setiap kiriman mencatat Nama Warga, Tanggal Update, Tinggi, dan Kualitas Pertumbuhan sehingga membentuk garis waktu bulanan.
  • 3. Warga cukup buka tautan website di HP tanpa lupa password atau membuat akun baru.

Standar Penilaian Kualitas

● Baik: Daun hijau sehat, batang kokoh, ada pertumbuhan tinggi/tunas.
● Cukup: Pertumbuhan melambat atau tanah agak kering, butuh pemupukan/penyiraman rutin.
● Perlu Perhatian: Terserang hama, daun menguning/layu, atau batang terganggu.

Kode Google Apps Script (Kode.gs)

Kode backend yang sudah disinkronkan dengan struktur kolom Log_Pohon dan upload foto Google Drive.

const FOLDER_ID = 'GANTI_DENGAN_ID_FOLDER_GOOGLE_DRIVE';
const SHEET_NAME = 'Log_Pohon';

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = getOrCreateSheet_(ss, SHEET_NAME);
    const values = sheet.getDataRange().getValues();

    if (values.length <= 1) {
      return jsonResponse_({ ok: true, data: [] });
    }

    const headers = values[0].map(function (h) {
      return String(h || '').trim();
    });

    const rows = [];
    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      const hasContent = row.some(function (cell) {
        return cell !== '' && cell !== null;
      });
      if (!hasContent) continue;

      const obj = {};
      for (let j = 0; j < headers.length; j++) {
        obj[headers[j]] = row[j];
      }
      rows.push(obj);
    }

    return jsonResponse_({ ok: true, data: rows });
  } catch (err) {
    return jsonResponse_({ ok: false, error: err.message });
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse_({ ok: false, error: 'Payload kosong' });
    }

    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = getOrCreateSheet_(ss, SHEET_NAME);

    let fotoUrl = String(payload.fotoUrl || payload.Foto_URL || '').trim();

    if (payload.fotoBase64 && String(payload.fotoBase64).indexOf('base64,') !== -1) {
      fotoUrl = saveBase64ToDrive_(
        payload.fotoBase64,
        payload.kodePohon || payload.Kode_Pohon || 'POHON',
        payload.namaFile || 'foto.jpg'
      );
    }

    const now = new Date();
    const bulanLaporan =
      payload.bulanLaporan ||
      payload.Bulan_Laporan ||
      Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM');

    const rowData = [
      now,
      String(payload.kodePohon || payload.Kode_Pohon || '').trim(),
      String(payload.namaWarga || payload.Nama_Warga || '').trim(),
      String(payload.wilayah || payload.Wilayah || '').trim(),
      String(payload.jenisPohon || payload.Jenis_Pohon || '').trim(),
      Number(payload.latitude || payload.Latitude || 0) || '',
      Number(payload.longitude || payload.Longitude || 0) || '',
      Number(payload.tinggiCm || payload.Tinggi_cm || 0) || '',
      String(payload.kualitas || payload.Kualitas || 'Baik').trim(),
      String(payload.catatan || payload.Catatan || '').trim(),
      fotoUrl,
      String(bulanLaporan).trim()
    ];

    sheet.appendRow(rowData);

    return jsonResponse_({
      ok: true,
      message: 'Data berhasil disimpan',
      fotoUrl: fotoUrl
    });
  } catch (err) {
    return jsonResponse_({ ok: false, error: err.message });
  }
}

Kode Google Apps Script (Sheet + Drive)

Salin kode ini ke Extensions → Apps Script di Google Sheet Anda.

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = getOrCreateSheet_(ss, "Pohon_Updates");
  const rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return jsonResponse_({ ok: true, records: [] });
  }
  const headers = rows[0];
  const records = rows.slice(1).map((r) => {
    const obj = {};
    headers.forEach((h, idx) => { obj[h] = r[idx]; });
    return obj;
  });
  return jsonResponse_({ ok: true, records: records });
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = getOrCreateSheet_(ss, "Pohon_Updates");

    let photoUrl = payload.photoDriveUrl || "";

    // Jika warga mengirim foto base64, simpan otomatis ke Google Drive
    if (payload.photoBase64 && payload.photoFileName) {
      const folderId = payload.gdriveFolderId || "";
      const folder = folderId
        ? DriveApp.getFolderById(folderId)
        : getOrCreateFolder_("Pantau_Pohon_Desa_Foto");

      const parts = payload.photoBase64.split(",");
      const mimeMatch = parts[0].match(/data:(.*);base64/);
      const mimeType = mimeMatch ? mimeMatch[1] : "image/jpeg";
      const decoded = Utilities.base64Decode(parts[1]);
      const blob = Utilities.newBlob(decoded, mimeType, payload.treeCode + "_" + payload.photoFileName);
      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Access.VIEW);
      photoUrl = "https://drive.google.com/uc?export=view&id=" + file.getId();
    }

    const row = [
      new Date().toISOString(),
      payload.treeCode || "",
      payload.species || "",
      payload.dusun || "",
      payload.lat || "",
      payload.lng || "",
      payload.plantedDate || "",
      payload.caretakerName || "",
      payload.updateDate || "",
      Number(payload.heightCm || 0),
      payload.trunkCm || "",
      payload.growthQuality || "Baik",
      payload.conditionNotes || "",
      photoUrl
    ];

    sheet.appendRow(row);
    return jsonResponse_({ ok: true, photoUrl: photoUrl });
  } catch (err) {
    return jsonResponse_({ ok: false, error: err.message });
  }
}

function getOrCreateSheet_(ss, name) {
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.appendRow([
      "timestamp_server", "treeCode", "species", "dusun", "lat", "lng",
      "plantedDate", "caretakerName", "updateDate", "heightCm", "trunkCm",
      "growthQuality", "conditionNotes", "photoDriveUrl"
    ]);
  }
  return sh;
}

function getOrCreateFolder_(name) {
  const iter = DriveApp.getFoldersByName(name);
  return iter.hasNext() ? iter.next() : DriveApp.createFolder(name);
}

function jsonResponse_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
  1. Buat Google Sheet baru di akun Google desa/pengelola, lalu buka menu Extensions > Apps Script.
  2. Tempelkan kode di atas, klik Save, lalu klik Deploy > New deployment (pilih tipe Web app, Execute as: Me, Who has access: Anyone).
  3. Salin URL Web App yang muncul dan tempelkan ke kolom di atas.

Hosting Gratis & Minim Perawatan

Karena aplikasi ini berupa satu berkas HTML statis yang membaca/menulis ke Google Sheet, biaya server bulanan bisa Rp 0 dan sangat awet:

1. GitHub Pages (Paling Populer & Awet)

Simpan berkas index.html di repositori publik GitHub, aktifkan GitHub Pages. Gratis selamanya untuk situs statis, HTTPS otomatis.

2. Cloudflare Pages

CDN global cepat, gratis, tanpa perlu kelola server/VPS, dan mudah dihubungkan dengan domain desa bila suatu saat tersedia.

3. Tips Minim Perawatan

Tidak ada database berbayar atau pembaruan server rutin. Semua catatan warga aman di Google Sheet dan foto terarsip rapi di Google Drive.
